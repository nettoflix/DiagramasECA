#     linha: 3        linha da grade dentro da fase (1 = logo abaixo do título).
#                     Sem "linha:", vai para a linha seguinte à da disciplina

#                     as linhas desenhadas em files/saved.txt.
#     pre: A, B       pré-requisitos (códigos, separados por vírgula)
#     grupos: a, b    grupos temáticos: informatica, controle, automacao,
#                     mecanica, eletrica, fisica_calculo (dois = caixa dividida)
#     preCH: 2300     horas-aula obrigatórias concluídas exigidas ("Pré CH")
#     optativa        não conta para as horas obrigatórias
